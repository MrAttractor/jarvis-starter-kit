function showPage(page) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.mobile-tab').forEach(t => t.classList.remove('active'));
  document.getElementById('page-' + page).classList.add('active');
  document.querySelectorAll('.nav-tab').forEach(t => {
    if (t.textContent.toLowerCase().includes(page.substring(0,4).toLowerCase()) ||
        (page === 'mbp' && t.textContent.includes('Plan')) ||
        (page === 'dashboard' && t.textContent.includes('Dashboard'))) {
      t.classList.add('active');
    }
  });
  document.querySelectorAll('.mobile-tab').forEach(t => {
    const icon = t.querySelector('.mobile-tab-icon').textContent;
    const map = { dashboard:'📊', finances:'💰', stack:'🛠️', actions:'🚀', taches:'✅' };
    if (map[page] === icon) t.classList.add('active');
  });
  if (page === 'dashboard') updateDashboard();
  if (page === 'finances') renderTransactions();
  if (page === 'stack') renderStack();
  if (page === 'actions') renderActions();
  if (page === 'taches') renderTaches();
  if (page === 'mbp') updateMBP();
}

  document.getElementById('nav-date').textContent = now.toLocaleDateString('fr-FR', opts);
  const today = now.toISOString().split('T')[0];
  document.querySelectorAll('input[type="date"]').forEach(d => { if (!d.value) d.value = today; });
}

// ============ FINANCES ============
function getMoisActuel() {
  const now = new Date();
  return `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}`;
}

function getMoisLabel() {
  const now = new Date();
  return now.toLocaleDateString('fr-FR', { month:'long', year:'numeric' });
}

function getCA() {
  const mois = getMoisActuel();
  return state.transactions
    .filter(t => t.type === 'entree' && t.date.startsWith(mois))
    .reduce((s, t) => s + (t.devise === 'XOF' ? t.montant/655 : t.montant), 0);
}

function getChargesStack() {
  return state.stack.reduce((s, t) => s + (t.cycle === 'annuel' ? t.cout/12 : t.cout), 0);
}

function ajouterTransaction() {
  const type = document.getElementById('tx-type').value;
  const montant = parseFloat(document.getElementById('tx-montant').value);
  const date = document.getElementById('tx-date').value;
  const desc = document.getElementById('tx-desc').value.trim();
  const client = document.getElementById('tx-client').value.trim();
  const devise = document.getElementById('tx-devise').value;
  const cat = document.getElementById('tx-categorie').value;

  if (!montant || !date || !desc) { alert('Montant, date et description requis.'); return; }

  const tx = {
    id: Date.now(),
    type, montant, date, desc, client, devise, cat,
    mois: date.substring(0,7)
  };

  state.transactions.unshift(tx);
  save();
  resetTxForm();
  renderTransactions();
  updateDashboard();
}

function resetTxForm() {
  document.getElementById('tx-montant').value = '';
  document.getElementById('tx-desc').value = '';
  document.getElementById('tx-client').value = '';
}

function toggleTxType() {
  const type = document.getElementById('tx-type').value;
  document.getElementById('tx-client-wrap').style.display = type === 'entree' ? 'block' : 'none';
  const cats = document.getElementById('tx-categorie');
  if (type === 'sortie') {
    cats.innerHTML = '<option value="outil">Outil / Stack</option><option value="marketing">Marketing</option><option value="formation">Formation</option><option value="autre">Autre</option>';
  } else {
    cats.innerHTML = '<option value="consulting">Consulting</option><option value="site-ia">Site IA Pro</option><option value="chatbot">Chatbot</option><option value="autre">Autre</option>';
  }
}

function renderTransactions() {
  const filterMois = document.getElementById('filter-mois')?.value || 'all';
  const filterType = document.getElementById('filter-type')?.value || 'all';

  // Populate mois filter
  const moisSet = [...new Set(state.transactions.map(t => t.mois))].sort().reverse();
  const moisEl = document.getElementById('filter-mois');
  if (moisEl) {
    const current = moisEl.value;
    moisEl.innerHTML = '<option value="all">Tous les mois</option>' +
      moisSet.map(m => `<option value="${m}" ${m===current?'selected':''}>${m}</option>`).join('');
  }

  let txs = [...state.transactions];
  if (filterMois !== 'all') txs = txs.filter(t => t.mois === filterMois);
  if (filterType !== 'all') txs = txs.filter(t => t.type === filterType);

  const container = document.getElementById('tx-list');
  if (!txs.length) { container.innerHTML = '<div class="empty"><div class="empty-icon">📭</div>Aucune transaction</div>'; }
  else {
    container.innerHTML = `<table class="table">
      <thead><tr><th>Date</th><th>Description</th><th>Catégorie</th><th>Montant</th><th></th></tr></thead>
      <tbody>${txs.map(t => `<tr>
        <td style="color:var(--text-muted);white-space:nowrap">${new Date(t.date).toLocaleDateString('fr-FR')}</td>
        <td>${t.desc}${t.client ? '<br><span style="font-size:0.7rem;color:var(--text-muted)">'+t.client+'</span>' : ''}</td>
        <td><span style="font-size:0.7rem;background:var(--card2);padding:2px 8px;border-radius:10px">${t.cat}</span></td>
        <td class="amount ${t.type==='entree'?'positif':'negatif'}">${t.type==='entree'?'+':'-'}${t.montant.toLocaleString('fr-FR')} ${t.devise==='XOF'?'FCFA':'€'}</td>
        <td><button onclick="supprimerTx(${t.id})" style="background:transparent;border:none;color:var(--text-muted);cursor:pointer;font-size:0.8rem">✕</button></td>
      </tr>`).join('')}</tbody>
    </table>`;
  }

  // Update résumé
  const moisLabel = document.getElementById('mois-label');
  if (moisLabel) moisLabel.textContent = getMoisLabel();

  const moisTxs = state.transactions.filter(t => t.mois === getMoisActuel());
  const entrees = moisTxs.filter(t => t.type==='entree').reduce((s,t) => s + (t.devise==='XOF'?t.montant/655:t.montant), 0);
  const sorties = moisTxs.filter(t => t.type==='sortie').reduce((s,t) => s + (t.devise==='XOF'?t.montant/655:t.montant), 0);
  const urssaf = entrees * 0.22;
  const dispo = entrees - sorties - urssaf;
  const charges = getChargesStack();
  const objectif = charges * 3;

  const fmt = n => n.toLocaleString('fr-FR', {maximumFractionDigits:0}) + ' €';

  document.getElementById('sum-entrees').textContent = fmt(entrees);
  document.getElementById('sum-sorties').textContent = '-' + fmt(sorties);
  document.getElementById('sum-solde').textContent = fmt(entrees - sorties);
  document.getElementById('sum-urssaf').textContent = fmt(urssaf);
  document.getElementById('sum-dispo').textContent = fmt(Math.max(0, dispo));

  const pctCA = objectif > 0 ? Math.min(100, (entrees/objectif)*100) : 0;
  const pctCharges = entrees > 0 ? Math.min(100, (sorties/entrees)*100) : 0;
  document.getElementById('prog-ca-fill').style.width = pctCA + '%';
  document.getElementById('prog-ca-pct').textContent = Math.round(pctCA) + '%';
  document.getElementById('prog-charges-fill').style.width = pctCharges + '%';
  document.getElementById('prog-charges-pct').textContent = Math.round(pctCharges) + '%';
}

function supprimerTx(id) {
  state.transactions = state.transactions.filter(t => t.id !== id);
  save();
  renderTransactions();
  updateDashboard();
}

// ============ DASHBOARD ============
function updateDashboard() {
  const ca = getCA();
  const charges = getChargesStack();
  const objectif = charges * 3;
  const urssaf = ca * 0.22;
  const reinvest = ca * 0.15;
  const treso = ca * 0.10;
  const net = ca * 0.53;
  const dispo = ca - urssaf;

  const fmt = n => n.toLocaleString('fr-FR', {maximumFractionDigits:0}) + ' €';
  const pct = objectif > 0 ? Math.min(100, (ca/objectif)*100) : 0;

  document.getElementById('kpi-ca').textContent = fmt(ca);
  document.getElementById('kpi-obj').textContent = fmt(objectif);
  document.getElementById('bar-ca').style.width = pct + '%';
  document.getElementById('kpi-dispo').textContent = fmt(Math.max(0, dispo));
  document.getElementById('bar-dispo').style.width = (ca > 0 ? Math.min(100, (dispo/ca)*100) : 0) + '%';
  document.getElementById('kpi-charges').textContent = fmt(charges);
  document.getElementById('kpi-urssaf').textContent = fmt(urssaf);

  document.getElementById('r-urssaf').textContent = fmt(urssaf);
  document.getElementById('r-reinvest').textContent = fmt(reinvest);
  document.getElementById('r-treso').textContent = fmt(treso);
  document.getElementById('r-net').textContent = fmt(net);

  // Dernières transactions dashboard
  const recent = state.transactions.slice(0, 4);
  const dashTx = document.getElementById('dash-transactions');
  if (!recent.length) { dashTx.innerHTML = '<div class="empty"><div class="empty-icon">📭</div>Aucune transaction</div>'; }
  else {
    dashTx.innerHTML = recent.map(t => `<div style="display:flex;justify-content:space-between;align-items:center;padding:10px 0;border-bottom:1px solid var(--border)">
      <div>
        <div style="font-size:0.82rem">${t.desc}</div>
        <div style="font-size:0.68rem;color:var(--text-muted)">${new Date(t.date).toLocaleDateString('fr-FR')} · ${t.cat}</div>
      </div>
      <div class="amount ${t.type==='entree'?'positif':'negatif'}" style="font-family:'DM Mono',monospace;font-size:0.82rem">${t.type==='entree'?'+':'-'}${t.montant} ${t.devise==='XOF'?'FCFA':'€'}</div>
    </div>`).join('');
  }

  // Prévision
  const joursDuMois = new Date(new Date().getFullYear(), new Date().getMonth()+1, 0).getDate();
  const jourActuel = new Date().getDate();
  const projectionCA = jourActuel > 0 ? (ca / jourActuel) * joursDuMois : 0;
  const prevEl = document.getElementById('prevision-content');
  const prevBadge = document.getElementById('prevision-badge');

  if (ca === 0) {
    prevEl.innerHTML = '<div class="alert info"><div class="alert-icon">📊</div><div>Enregistre tes premières transactions pour activer les prévisions.</div></div>';
    prevBadge.textContent = 'En attente';
  } else {
    const onTrack = projectionCA >= objectif;
    prevBadge.textContent = onTrack ? '✅ En bonne voie' : '⚠️ En dessous';
    prevBadge.className = 'card-badge ' + (onTrack ? 'badge-vert' : 'badge-rouge');
    prevEl.innerHTML = `<div class="alert ${onTrack?'success':'danger'}">
      <div class="alert-icon">${onTrack?'📈':'⚠️'}</div>
      <div>Projection fin de mois : <strong>${fmt(projectionCA)}</strong> (objectif : ${fmt(objectif)}). 
      ${onTrack ? 'Tu es sur la bonne trajectoire !' : 'Il te faut '+fmt(objectif-ca)+' supplémentaires pour atteindre l\'objectif.'}</div>
    </div>`;
  }

  // Tâches dashboard
  const urgentes = state.taches.filter(t => !t.done && t.priority === 'high').slice(0, 4);
  const dashT = document.getElementById('dash-taches');
  if (!urgentes.length) { dashT.innerHTML = '<div class="empty"><div class="empty-icon">🎉</div>Aucune tâche urgente</div>'; }
  else {
    dashT.innerHTML = urgentes.map(t => `<div style="display:flex;align-items:center;gap:10px;padding:10px 0;border-bottom:1px solid var(--border)">
      <div class="tache-check ${t.done?'done':''}" onclick="toggleTache(${t.id})"></div>
      <div style="flex:1;font-size:0.82rem;${t.done?'text-decoration:line-through;color:var(--text-muted)':''}">${t.titre}</div>
      <span class="tache-priority priority-high">URGENT</span>
    </div>`).join('');
  }
}

// ============ STACK ============
function renderStack() {
  const list = document.getElementById('stack-list');
  const total = getChargesStack();
  document.getElementById('stack-total').textContent = total.toLocaleString('fr-FR', {maximumFractionDigits:0}) + ' €';
  document.getElementById('stack-total-badge').textContent = total.toLocaleString('fr-FR', {maximumFractionDigits:0}) + ' €/mois';

  list.innerHTML = state.stack.map(s => {
    const cout = s.cycle === 'annuel' ? (s.cout/12).toFixed(0) + ' €/mois (' + s.cout + '€/an)' : s.cout + ' €/mois';
    const today = new Date();
    let renewalClass = '';
    let renewalText = s.renewal || 'Date non renseignée';
    if (s.renewal) {
      const diff = Math.ceil((new Date(s.renewal) - today) / (1000*60*60*24));
      if (diff <= 30) renewalClass = 'soon';
      renewalText = 'Renouvellement dans ' + diff + ' jours';
    }
    return `<div class="stack-item">
      <div class="stack-left">
        <div class="stack-icon" style="background:var(--or-dim)">${s.icon||'🔧'}</div>
        <div>
          <div class="stack-name">${s.nom}</div>
          <div class="stack-cycle">${s.cycle}</div>
        </div>
      </div>
      <div class="stack-right">
        <div class="stack-cost">-${cout}</div>
        <div class="stack-renewal ${renewalClass}">${renewalText}</div>
      </div>
    </div>`;
  }).join('');
}

function ajouterStack() {
  const nom = document.getElementById('stack-nom').value.trim();
  const cout = parseFloat(document.getElementById('stack-cout').value);
  const cycle = document.getElementById('stack-cycle').value;
  const renewal = document.getElementById('stack-renewal').value;
  const icon = document.getElementById('stack-icon').value || '🔧';
  if (!nom || !cout) { alert('Nom et coût requis.'); return; }
  state.stack.push({ id: Date.now(), nom, cout, cycle, renewal, icon });
  save();
  renderStack();
  document.getElementById('stack-nom').value = '';
  document.getElementById('stack-cout').value = '';
  document.getElementById('stack-icon').value = '';
}

let tacheFilter = 'all';

function ajouterTache() {
  const titre = document.getElementById('tache-titre').value.trim();
  const priority = document.getElementById('tache-priority').value;
  const date = document.getElementById('tache-date').value;
  const cat = document.getElementById('tache-cat').value;
  if (!titre) { alert('Titre requis.'); return; }
  state.taches.unshift({ id: Date.now(), titre, priority, date, cat, done: false });
  save();
  renderTaches();
  document.getElementById('tache-titre').value = '';
}

function toggleTache(id) {
  const t = state.taches.find(t => t.id === id);
  if (t) { t.done = !t.done; save(); renderTaches(); updateDashboard(); }
}

function filterTaches(f) { tacheFilter = f; renderTaches(); }

function renderTaches() {
  let taches = [...state.taches];
  if (tacheFilter === 'pending') taches = taches.filter(t => !t.done);
  if (tacheFilter === 'done') taches = taches.filter(t => t.done);

  const total = state.taches.length;
  const done = state.taches.filter(t => t.done).length;
  const pct = total > 0 ? Math.round((done/total)*100) : 0;

  document.getElementById('taches-pct').textContent = pct + '%';
  document.getElementById('taches-fill').style.width = pct + '%';
  document.getElementById('count-high').textContent = state.taches.filter(t => t.priority==='high' && !t.done).length;
  document.getElementById('count-mid').textContent = state.taches.filter(t => t.priority==='mid' && !t.done).length;
  document.getElementById('count-done').textContent = done;

  const list = document.getElementById('taches-list');
  if (!taches.length) { list.innerHTML = '<div class="empty"><div class="empty-icon">✅</div>Aucune tâche ici</div>'; return; }

  const priorityClass = { high:'priority-high', mid:'priority-mid', low:'priority-low' };
  const priorityLabel = { high:'URGENT', mid:'MOYEN', low:'BAS' };

  list.innerHTML = taches.map(t => `<div class="tache-item">
    <div class="tache-check ${t.done?'done':''}" onclick="toggleTache(${t.id})"></div>
    <div class="tache-content">
      <div class="tache-title ${t.done?'done':''}">${t.titre}</div>
      <div class="tache-meta">${t.cat}${t.date?' · '+new Date(t.date).toLocaleDateString('fr-FR'):''}</div>
    </div>
    <span class="tache-priority ${priorityClass[t.priority]}">${priorityLabel[t.priority]}</span>
    <button onclick="supprimerTache(${t.id})" style="background:transparent;border:none;color:var(--text-muted);cursor:pointer;font-size:0.8rem;flex-shrink:0">✕</button>
  </div>`).join('');
}

function supprimerTache(id) {
  state.taches = state.taches.filter(t => t.id !== id);
  save();
  renderTaches();
}
