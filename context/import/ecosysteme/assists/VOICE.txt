<script>
  // ============================================================
  // INTÉGRATION — Voici les 3 points à brancher sur TON process existant.
  // ============================================================

  const chat = document.getElementById("chat");
  const input = document.getElementById("input");
  const interim = document.getElementById("interim");
  const mic = document.getElementById("mic");
  const send = document.getElementById("send");

  function bulle(texte, qui){
    const d = document.createElement("div");
    d.className = "msg " + qui;
    if(qui==="a"){ d.innerHTML = `<span class="nom">Monsieur S.</span>${escapeHtml(texte)}`; }
    else { d.textContent = texte; }
    chat.appendChild(d);
    chat.scrollTop = chat.scrollHeight;
  }
  function escapeHtml(s){return String(s).replace(/[&<>"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;"}[c]));}

  // -------- POINT D'INTÉGRATION 1 : envoi d'un message --------
  // Remplace le corps de cette fonction par TON appel existant
  // (ton fetch GET vers Apps Script, ou ton client Supabase).
  function envoyerMessage(texte){
    if(!texte || !texte.trim()) return;
    bulle(texte.trim(), "u");        // affiche le message utilisateur
    input.value = "";
    interim.textContent = "";

    // >>> ICI : appelle ton backend Attractor Assists existant <<<
    // ex: fetch(`${APPS_SCRIPT_URL}?action=chat&clientId=${id}&msg=${encodeURIComponent(texte)}`)
    //       .then(r=>r.json()).then(d=> bulle(d.reponse, "a"));
    reponseDemo(texte); // <-- à SUPPRIMER une fois ton backend branché
  }

  // -------- POINT D'INTÉGRATION 2 : le micro --------
  const voix = new AttractorVoice({
    lang: "fr-FR",
    autoStopMs: 1800,
    onState: (s) => {
      if(s==="listening"){ mic.classList.add("live"); document.getElementById("hint").textContent="J'écoute… parle."; }
      else { mic.classList.remove("live"); document.getElementById("hint").textContent="Appuie, parle librement. L'assistant s'occupe du reste."; }
    },
    onInterim: (txt) => { interim.textContent = txt; },   // texte en cours, live
    onFinal: (txt) => {                                    // texte final
      interim.textContent = "";
      input.value = txt;
      // Effet "je parle, il fait" : envoi automatique.
      // Mets autoSend=false si tu préfères laisser l'utilisateur relire/valider.
      const autoSend = true;
      if(autoSend) envoyerMessage(txt);
    },
    onError: (e) => {
      if(e.type==="not-allowed"){ document.getElementById("hint").textContent="Autorise le micro dans ton navigateur."; }
      else if(e.type==="unsupported"){ showUnsupported(); }
    },
  });

  if(!voix.isSupported()){ showUnsupported(); }
  function showUnsupported(){ document.getElementById("unsupported").style.display="block"; }

  mic.addEventListener("click", () => voix.toggle());
  send.addEventListener("click", () => envoyerMessage(input.value));
  input.addEventListener("keydown", (e)=>{ if(e.key==="Enter" && !e.shiftKey){ e.preventDefault(); envoyerMessage(input.value);} });

  // -------- POINT D'INTÉGRATION 3 : le 1er message de l'assistant --------
  // Remplace par ton onboarding réel (warm welcome → nickname → …)
  bulle("Bonjour ! Dis-moi tout ce que tu as en tête en ce moment. Ne structure pas, parle simplement — je m'occupe d'y voir clair avec toi.", "a");

  // ===== Démo locale (à retirer en prod) =====
  function reponseDemo(texte){
    setTimeout(()=>{
      bulle("Bien reçu. J'ai noté l'essentiel. On va dégager ensemble ton couloir prioritaire — celui qui te rapproche du #1 dans ton domaine.", "a");
    }, 700);
  }
</script>
