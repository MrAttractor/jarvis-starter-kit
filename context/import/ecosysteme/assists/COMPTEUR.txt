<script>
  // ====== CONFIG ======
  const SUPABASE_URL="https://<PROJECT_REF>.supabase.co";
  const SUPABASE_ANON_KEY="<SUPABASE_ANON_KEY>";
  const REST=`${SUPABASE_URL}/rest/v1`;
  const HEADERS={apikey:SUPABASE_ANON_KEY,Authorization:`Bearer ${SUPABASE_ANON_KEY}`,"Content-Type":"application/json"};

  const toastEl=document.getElementById("toast");
  function toast(m){toastEl.textContent=m;toastEl.classList.add("show");setTimeout(()=>toastEl.classList.remove("show"),1800);}
  const eur=n=>Number(n||0).toLocaleString("fr-FR",{maximumFractionDigits:0});
  function esc(s){return String(s??"").replace(/[&<>"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;"}[c]));}

  let CONFIG=null;

  async function chargerConfig(){
    const r=await fetch(`${REST}/config_finance?id=eq.1`,{headers:HEADERS});
    const d=await r.json(); CONFIG=(d&&d[0])||null;
  }

  async function chargerSynthese(){
    // mois en cours
    const moisCourant=new Date(); moisCourant.setDate(1);
    const moisISO=moisCourant.toISOString().slice(0,10);
    const r=await fetch(`${REST}/synthese_mensuelle?mois=eq.${moisISO}`,{headers:HEADERS});
    const d=await r.json(); const m=(d&&d[0])||null;

    const obj=CONFIG?Number(CONFIG.objectif_actuel):10000;
    const ca=m?Number(m.ca_total):0;
    const pct=Math.min(100,Math.round(ca/obj*100));
    document.getElementById("caMois").innerHTML=`${eur(ca)} <small>€</small>`;
    document.getElementById("cible").textContent=`Objectif : ${eur(obj)} € · cible ${CONFIG?CONFIG.date_cible:"—"}`;
    document.getElementById("barFill").style.width=pct+"%";
    document.getElementById("barPct").textContent=pct+"% de l'objectif";

    document.getElementById("sProd").textContent=eur(m?m.ca_produits:0)+" €";
    document.getElementById("sMrr").textContent=eur(m?m.ca_mrr:0)+" €";
    document.getElementById("sCharges").textContent=eur(m?m.part_charges:0)+" €";
    document.getElementById("sCamp").textContent=eur(m?m.part_campagne:0)+" €";
    document.getElementById("sMarge").textContent=eur(m?m.part_marge:0)+" €";
    const mrrPct=obj>0?Math.round((m?m.ca_mrr:0)/obj*100):0;
    document.getElementById("sMrrPct").textContent=mrrPct+" %";
  }

  async function chargerStabilite(){
    try{
      const r=await fetch(`${REST}/statut_stabilite`,{headers:HEADERS});
      const d=await r.json(); const s=(d&&d[0])||null;
      const box=document.getElementById("stabBox");
      const title=document.getElementById("stabTitle");
      const msg=document.getElementById("stabMsg");
      const moisDiv=document.getElementById("stabMois");
      if(!s){msg.textContent="Pas encore de données.";return;}

      if(s.stabilise){
        box.className="stab ok"; title.textContent="✦ Palier stabilisé";
        msg.innerHTML=`Tu peux déclencher le palier Fibonacci suivant. Conditions réunies sur ${s.mois_requis} mois.`;
      }else{
        box.className="stab no"; title.textContent="Stabilité en cours";
        msg.innerHTML=`Objectif : ${s.mois_requis} mois consécutifs avec CA ≥ palier, marge positive, et MRR suffisant. ${s.mois_evalues}/${s.mois_requis} mois évalués.`;
      }
      const detail=Array.isArray(s.detail)?s.detail:[];
      moisDiv.innerHTML=detail.map(x=>`
        <div class="m ${x.ok?'good':''}">
          <div class="dot">${x.ok?'✓':'·'}</div>
          ${new Date(x.mois).toLocaleDateString("fr-FR",{month:'short'})}<br>${eur(x.ca)}€
        </div>`).join("")||"";
    }catch(e){/* silencieux */}
  }

  async function chargerListe(){
    const r=await fetch(`${REST}/encaissements?order=date_encaiss.desc&limit=15`,{headers:HEADERS});
    const d=await r.json()||[];
    document.getElementById("listeEnc").innerHTML=d.map(e=>`
      <div class="row">
        <div class="lib">${esc(e.libelle)}<span class="tag ${e.type}">${e.type}</span></div>
        <div class="mt">${eur(e.montant)} €</div>
      </div>`).join("")||`<div style="color:#8A8A8E;font-family:'Helvetica Neue',sans-serif;font-size:13px;padding:10px 2px">Aucun encaissement enregistré.</div>`;
  }

  async function ajouter(){
    const libelle=document.getElementById("eLib").value.trim();
    const montant=parseFloat(document.getElementById("eMontant").value);
    const type=document.getElementById("eType").value;
    const pilier=document.getElementById("ePilier").value;
    if(!libelle||!montant){toast("Libellé + montant requis");return;}
    try{
      await fetch(`${REST}/encaissements`,{method:"POST",
        headers:{...HEADERS,"Prefer":"return=minimal"},
        body:JSON.stringify({libelle,montant,type,pilier})});
      document.getElementById("eLib").value="";document.getElementById("eMontant").value="";
      document.getElementById("form").classList.remove("open");
      toast("Encaissement enregistré");
      await tout();
    }catch(e){toast("Erreur réseau");}
  }

  async function tout(){ await chargerConfig(); await chargerSynthese(); await chargerStabilite(); await chargerListe(); }

  document.getElementById("addToggle").addEventListener("click",()=>document.getElementById("form").classList.toggle("open"));
  tout();
  setInterval(tout,45000);
</script>
