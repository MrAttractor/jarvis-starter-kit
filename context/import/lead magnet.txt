// ═══ INSCRIPTION PROSPECT ═══
function inscrireProspectLeadMagnet(prenom, email, whatsapp, secteur) {
  const ss = SpreadsheetApp.openById(CONFIG.MASTER_SHEET_ID);
  let sheet = ss.getSheetByName(CONFIG.ONGLET_LEAD_MAGNET);
  
  if (!sheet) {
    Logger.log('❌ ERREUR: Onglet LEAD_MAGNET non trouvé');
    return;
  }
  
  const date = new Date();
  const dateStr = Utilities.formatDate(date, 'GMT+1', 'yyyyMMdd');
  const random = Math.floor(Math.random() * 100);
  const id = `LM-${dateStr}-${prenom.substring(0,1).toUpperCase()}${random}`;
  
  sheet.appendRow([date, id, prenom, email, whatsapp, secteur, 'Inscrit', 0, 0, '', '', '', 'Inscrit via formulaire']);
  
  Logger.log(`✅ Prospect ${prenom} inscrit avec ID ${id}`);
  
  envoyerEmail('J0_LIVRAISON', prenom, email, id, sheet.getLastRow());
  
  return id;
}

// ═══ ENVOI QUOTIDIEN AUTOMATIQUE ═══
function envoyerEmailsQuotidiens() {
  const sheet = SpreadsheetApp.openById(CONFIG.MASTER_SHEET_ID).getSheetByName(CONFIG.ONGLET_LEAD_MAGNET);
  const data = sheet.getDataRange().getValues();
  const aujourdhui = new Date();
  let emailsEnvoyes = 0;
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const dateInscription = new Date(row[0]);
    const id = row[1];
    const prenom = row[2];
    const email = row[3];
    const statut = row[6];
    const etapeChallenge = row[7] || 0;
    
    if (statut === 'Converti' || statut === 'Abandonné') continue;
    
    const joursDepuis = Math.floor((aujourdhui - dateInscription) / (1000 * 60 * 60 * 24));
    const emailType = determinerEmailAEnvoyer(joursDepuis, etapeChallenge);
    
    if (emailType) {
      envoyerEmail(emailType, prenom, email, id, i + 1);
      emailsEnvoyes++;
    }
  }
  
  Logger.log(`✅ ${emailsEnvoyes} email(s) envoyé(s)`);
}

// ═══ DÉTERMINER EMAIL ═══
function determinerEmailAEnvoyer(joursDepuis, etapeChallenge) {
  if (joursDepuis === 1 && etapeChallenge === 0) return 'J1_QUALIFICATION';
  if (joursDepuis === 2 && etapeChallenge === 0) return 'J2_ANNONCE';
  if (joursDepuis === 3 && etapeChallenge < 1) return 'CHALLENGE_J1';
  if (joursDepuis === 4 && etapeChallenge < 2) return 'CHALLENGE_J2';
  if (joursDepuis === 5 && etapeChallenge < 3) return 'CHALLENGE_J3';
  if (joursDepuis === 6 && etapeChallenge < 4) return 'CHALLENGE_J4';
  if (joursDepuis === 7 && etapeChallenge < 5) return 'CHALLENGE_J5';
  if (joursDepuis === 8 && etapeChallenge < 6) return 'CHALLENGE_J6';
  if (joursDepuis === 9 && etapeChallenge < 7) return 'CHALLENGE_J7';
  if (joursDepuis === 10 && etapeChallenge >= 7) return 'J10_FELICITATIONS';
  if (joursDepuis === 11 && etapeChallenge >= 7) return 'J11_CONVERSION';
  return null;
}

// ═══ ENVOYER EMAIL ═══
function envoyerEmail(typeEmail, prenom, emailDest, idProspect, ligneSheet) {
  const sujet = getEmailSubject(typeEmail);
  const corps = getEmailBody(typeEmail, prenom, idProspect);
  
  try {
    GmailApp.sendEmail(emailDest, sujet, '', {
      htmlBody: corps,
      name: CONFIG.PRENOM_EXPEDITEUR
    });
    
    const sheet = SpreadsheetApp.openById(CONFIG.MASTER_SHEET_ID).getSheetByName(CONFIG.ONGLET_LEAD_MAGNET);
    sheet.getRange(ligneSheet, 10).setValue(new Date());
    sheet.getRange(ligneSheet, 11).setValue(typeEmail);
    
    Logger.log(`✅ Email ${typeEmail} envoyé à ${prenom}`);
  } catch (error) {
    Logger.log(`❌ Erreur: ${error}`);
  }
}

// ═══ WEB APP — VALIDATION ═══
function doGet(e) {
  const idProspect = e.parameter.id;
  const jour = parseInt(e.parameter.jour) || 1;
  
  if (!idProspect) {
    return HtmlService.createHtmlOutput('<h1>Erreur : ID manquant</h1>');
  }
  
  const sheet = SpreadsheetApp.openById(CONFIG.MASTER_SHEET_ID).getSheetByName(CONFIG.ONGLET_LEAD_MAGNET);
  const data = sheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][1] === idProspect) {
      const prenom = data[i][2];
      sheet.getRange(i + 1, 8).setValue(jour);
      const nouvelleNote = `${data[i][12] || ''}\nJour ${jour} validé le ${new Date().toLocaleString('fr-FR')}`;
      sheet.getRange(i + 1, 13).setValue(nouvelleNote);
      return HtmlService.createHtmlOutput(genererPageConfirmation(prenom, jour));
    }
  }
  
  return HtmlService.createHtmlOutput('<h1>Prospect non trouvé</h1>');
}

// ═══ FORMULAIRE SITE ═══
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    if (data.action === 'inscrire') {
      inscrireProspectLeadMagnet(data.prenom, data.email, data.whatsapp, data.secteur);
      return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (error) {
    Logger.log('Erreur doPost: ' + error);
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
