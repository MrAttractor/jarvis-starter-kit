    showToast('📅 Demande envoyée ! +10 pts dès confirmation.');
    // Animate points

    const target = 350;
    const el = document.getElementById('pts-count');
    const interval = setInterval(() => {
      pts++;
      el.textContent = pts;
      if (pts >= target) clearInterval(interval);
    }, 50);
  }

  }

<script>
  let selectedColor = '#C9A84C';
  let clientId = '';

  function goStep(n) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById('screen-' + n).classList.add('active');
    document.getElementById('step-num').textContent = n;

    document.querySelectorAll('.step').forEach((s, i) => {
      s.classList.remove('active', 'done');
      if (i + 1 < n) s.classList.add('done');
      if (i + 1 === n) s.classList.add('active');
    });
  }

  function selectSecteur(el) {
    document.querySelectorAll('.secteur-btn').forEach(b => b.classList.remove('selected'));
    el.classList.add('selected');
  }

  function selectColor(el, color) {
    document.querySelectorAll('.color-swatch').forEach(s => s.classList.remove('selected'));
    el.classList.add('selected');
    selectedColor = color;
  }

  function handleLogoUpload(input) {
    if (input.files[0]) {
      const zone = input.closest('.upload-zone');
      zone.innerHTML = `<div class="upload-icon">✅</div><div class="upload-text"><strong>${input.files[0].name}</strong> importé</div>`;
    }
  }

  function generateId(prenom) {
    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
    let id = '';
    for (let i = 0; i < 5; i++) id += chars[Math.floor(Math.random() * chars.length)];
    return `FID-${prenom.charAt(0).toUpperCase() + prenom.slice(1).toLowerCase()}-${id}`;
  }

  function createAccount() {
    const prenom = document.getElementById('prenom').value || 'Client';
    clientId = generateId(prenom);

    document.getElementById('link-cockpit').textContent = `fidelys.agenceattractor.com/cockpit/${clientId}`;
    document.getElementById('link-client').textContent = `fidelys.agenceattractor.com/client/${clientId}`;

    const expiry = new Date();
    expiry.setDate(expiry.getDate() + 14);
    document.getElementById('expiry-date').textContent = expiry.toLocaleDateString('fr-FR', { day: 'numeric', month: 'long', year: 'numeric' });

    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById('screen-success').classList.add('active');
    document.getElementById('step-num').textContent = '✓';
    document.querySelectorAll('.step').forEach(s => { s.classList.remove('active'); s.classList.add('done'); });
  }

  function copyLink(id) {
    const text = document.getElementById(id).textContent;
    navigator.clipboard.writeText('https://' + text).then(() => {
      const btn = event.target;
      btn.textContent = '✓ Copié';
      btn.style.background = 'rgba(45,123,79,0.2)';
      btn.style.color = '#4CAF82';
      setTimeout(() => { btn.textContent = 'Copier'; btn.style.background = ''; btn.style.color = ''; }, 2000);
    });
  }

  function goToCockpit() {
    alert(`Redirection vers votre cockpit : ${clientId}\n\n(En production : ouverture du cockpit personnalisé)`);
  }
</script>
